cmake_minimum_required(VERSION 3.15)
project(ct_offline_replay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(ASMJIT_STATIC ON CACHE BOOL "" FORCE)
set(ASMJIT_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory(upstream/asmjit EXCLUDE_FROM_ALL)
file(GLOB base upstream/dsp56300/source/dsp56kBase/*.cpp)
list(FILTER base EXCLUDE REGEX "_test\\.cpp$")
file(GLOB core upstream/dsp56300/source/dsp56kEmu/*.cpp)
list(FILTER core EXCLUDE REGEX "(test|Test|dspthread|haltDSP)\\.cpp$")
# Exclude all test classes regardless of capitalization/prefix.
list(FILTER core EXCLUDE REGEX "[Tt]ests?\\.cpp$")
add_library(replay_core STATIC ${base} ${core})
target_include_directories(replay_core PUBLIC upstream/dsp56300/source)
target_compile_definitions(replay_core PUBLIC DSP56K_FORCE_INTERPRETER=1 DSP56300_DEBUGGER=0)
target_link_libraries(replay_core PUBLIC asmjit)
if(MSVC)
  target_compile_options(replay_core PRIVATE /constexpr:steps100000000 /bigobj /EHsc)
endif()
add_executable(ct_replay_fixture harness/main.cpp)
target_link_libraries(ct_replay_fixture PRIVATE replay_core)
